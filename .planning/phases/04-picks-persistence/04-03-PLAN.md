  <files>app/composables/usePickProgress.ts, tests/composables/usePickProgress.test.ts</files>
  <behavior>
    - Function `usePickProgress(season = 2026)` returns an object with:
      - `progressOverall: Computed<{ picked: number, total: number }>` — counts for entire season
      - `progressForWeek(weekNum: number): Computed<{ picked: number, total: number }>` — counts for a specific week
    - Progress is derived from `usePicksStorage()` (picks count) and `useGames()` (total games or week-filtered games)
    - Computed updates reactively whenever picks.value or games.value changes
    - `picked` = number of keys in picks.value
    - `total` = total games in season OR games in the specified week (weekNum parameter)
    - Edge cases: if games data is not yet loaded, total = 0 (badge shows "0/0 picked")
  </behavior>
  <action>
    Create `app/composables/usePickProgress.ts`:
    
    ```typescript
    import { usePicksStorage } from './usePicksStorage'
    import { useGames } from './useGames'
    
    export function usePickProgress(season = 2026) {
      const picks = usePicksStorage(season)
      const { data: gamesData } = useGames(season)
      
      const progressOverall = computed(() => {
        const games = gamesData.value?.games ?? []
        const total = games.length
        const picked = Object.keys(picks.value).length
        return { picked, total }
      })
      
      function progressForWeek(weekNum: number) {
        return computed(() => {
          const games = gamesData.value?.games ?? []
          const weekGames = games.filter(g => g.week === weekNum)
          const total = weekGames.length
          const picked = weekGames.filter(g => g.id in picks.value).length
          return { picked, total }
        })
      }
      
      return { progressOverall, progressForWeek }
    }
    ```
    
    This pattern avoids manual state: progress is always in sync with picks via computed dependencies.
  </action>
